# 面談まわりの改善 実装計画

> **作業する人へ：** この計画は `superpowers:subagent-driven-development` または
> `superpowers:executing-plans` で1タスクずつ進めることを前提にしています。
> 手順はチェックボックス（`- [ ]`）で追えるようにしてあります。

**目的：** 面談の申請（インターン生）と確定（スタッフ）まわりの6つの不具合・使いにくさを直す。

**設計書：** `_specs/2026-08-06-面談まわりの改善-design.md`（判断の理由はすべてこちら）

**進め方：** 表示の規則 → 説明文 → SSE のサーバー側 → 暗転しない再描画 → SSE の画面側 →
アコーディオン → 第N希望の色、の順。前のタスクが次のタスクの土台になるので順番を守る。

**使う技術：** Node標準機能のみ（Express・`@libsql/client` は既存）。**npm への追加は一切しない。**

- **料金が発生することは絶対にしない。** 外部サービスも npm の追加依存も入れない

   表示だけを受け持つ関数の検証

   使い方:  node tools/test-fmt.mjs

   apply.html と index.html は <script> 直書きなので import できない。

let pass = 0;

function check(label, actual, expected) {
  const ok = JSON.stringify(actual) === JSON.stringify(expected);

/* `function 名(` から、対応する閉じ波括弧までを切り出す */
function cut(src, name) {
  const start = src.indexOf(`function ${name}(`);
  if (start < 0) throw new Error(`${name} が見つかりません（名前を変えたならこのファイルも直すこと）`);
  let depth = 0;
  for (let j = src.indexOf('{', start); j < src.length; j++) {
    if (src[j] === '{') depth++;
    else if (src[j] === '}') { depth--; if (depth === 0) return src.slice(start, j + 1); }
  }
  throw new Error(`${name} の終わりが見つかりません`);
}
/* 指定した関数だけを取り出して、動かせる形にして返す */
function load(file, names, prelude = '') {
  const src = fs.readFileSync(path.join(ROOT, file), 'utf8');
  const body = names.map((n) => cut(src, n)).join('\n');
  return new Function(`${prelude}\n${body}\nreturn {${names.join(',')}};`)();
}

function iso(h, m, day = 7) {

```js
/* 区間の終わりに出す時刻。
   枠が2つ以上あるときは最後の枠の開始時刻、1つだけのときは開始＋30分。

   表の1行は「その時刻から30分の枠」だが、申請する人は行の見出しを読んで
   「17:00から18:00まで選んだ」と受け取る。その読み方に合わせている。
   面談そのものは30分なので、ここに出る終了時刻より30分あとに終わることがある
   （設計書 _specs/2026-08-06-面談まわりの改善-design.md の 1.4）*/
function runEnd(run){
  var last=run[run.length-1];
  if(run.length>1)return new Date(last);
  return new Date(new Date(last).getTime()+SLOT_MS);
}
/* 続いた枠を「9:00〜11:00」にまとめ、離れていれば ＆ でつなぐ */
function fmtRanges(list){
  var sorted=list.slice().sort(function(a,b){return new Date(a)-new Date(b);});
  var runs=[];
  sorted.forEach(function(iso){
    var run=runs[runs.length-1];
    var prev=run&&run[run.length-1];
    if(prev&&new Date(iso)-new Date(prev)===SLOT_MS)run.push(iso);
    else runs.push([iso]);
  });
  return runs.map(function(r){
    return hm(r[0])+'〜'+hm(runEnd(r));
  }).join('＆');
}
```

```js
/* 区間の終わりに出す時刻。理由と規則は apply.html の runEnd と同じ
   （設計書 _specs/2026-08-06-面談まわりの改善-design.md の 1.4）。
   申請画面とスタッフ画面で違う数え方をすると、同じ希望が別の時間に見えてしまう */
function runEnd(run){
  const last=run[run.length-1];
  if(run.length>1)return new Date(last);
  return new Date(new Date(last).getTime()+SLOT_MS);
}
/* まとまりの見出し。「8/6(木) 9:00〜11:00＆14:00〜22:30」。
   終日OKの日は途中が抜けることがあるので、続いた区間ごとに ＆ でつなぐ */
function fmtGroupRange(g){
  if(!g||!g.slots.length)return '—';
  const runs=[];
  g.slots.slice().sort((a,b)=>new Date(a)-new Date(b)).forEach(iso=>{
    const run=runs[runs.length-1];
    const prev=run&&run[run.length-1];
    if(prev&&new Date(iso)-new Date(prev)===SLOT_MS)run.push(iso);
    else runs.push([iso]);
  });
  const times=runs.map(r=>`${hm(r[0])}〜${hm(runEnd(r))}`).join('＆');
  return `${fmtDate(g.slots[0])} ${times}`;
}
```

```js
/* SSE につないで、届いた refresh を数える。
   fetch のストリームをそのまま読む（画面側と同じ受け方）。close() で切る */
async function openStream(token) {
  const ctrl = new AbortController();
  let res;
  try { res = await fetch(BASE + '/api/stream', { headers: H(token), signal: ctrl.signal }); }
  catch (e) { return { status: 0, events: [], close() { ctrl.abort(); } }; }
  const events = [];
  if (!res.ok) { ctrl.abort(); return { status: res.status, events, close() {} }; }
  (async () => {
    const reader = res.body.getReader();
    const dec = new TextDecoder();
    let buf = '';
    try {
      for (;;) {
        const { value, done } = await reader.read();
        if (done) break;
        buf += dec.decode(value, { stream: true });
        let i;
        while ((i = buf.indexOf('\n\n')) >= 0) {
          const block = buf.slice(0, i);
          buf = buf.slice(i + 2);
          if (block.startsWith('event: refresh')) events.push(block);
        }
      }
    } catch { /* close() で切ったとき。ここは正常 */ }
  })();
  return { status: res.status, events, close() { ctrl.abort(); } };
}
/* 条件が満たされるまで待つ。押し出しは非同期なので、決め打ちの待ち時間にしない */
async function waitFor(fn, ms = 3000) {
  const until = Date.now() + ms;
  while (Date.now() < until) {
    if (fn()) return true;
    await new Promise((r) => setTimeout(r, 50));
  }
  return false;
}
```

   つながっている画面へ「更新があった」ことを押し出す仕組み（SSE）

   面談の申請や確定は、これまで画面を再読み込みするまで気づけなかった。
   定期ポーリングが store の updatedAt しか見ておらず、

   ここでは中身を送らず「更新があった」ことだけを送る。

   受け取った画面が /api/db を取り直す。

   接続の一覧はこのプロセスのメモリに置く。
   Render 無料プランはインスタンスが1つなので成立する（readDB のキャッシュと同じ前提）。
   インスタンスを増やす日が来たら、ここは作り直しが必要。
   ========================================================= */

/* 無反応の接続は途中の経路で切られる。生きていることを知らせる間隔。
   Render の前段はおよそ100秒で切るので、その半分より短くしてある */
const HEARTBEAT_MS = 25000;

const clients = new Set();

/* つないできた画面を控える。戻り値をそのまま removeClient に渡す */
function addClient(res, user) {
  const c = {
    res,
    userId: user.id,
    branchId: user.branch_id || null,
    role: user.role,
  };
  clients.add(c);
  return c;
}
function removeClient(c) {
  clients.delete(c);
}

/* その通知がこの人に見えるか。
   server.js の listNotificationsFor と同じ規則にしてある。
   ここがずれると、届いた合図で取り直しても中身が増えず無駄になる */
function canSee(c, branchId) {
  if (c.role === 'admin') return true;
  if (branchId == null) return true;
  return c.branchId === branchId;
}

/* 見える人にだけ押し出す。書けない相手は切れているので一覧から外す */
function broadcast(branchId, payload) {
  const body = `event: refresh\ndata: ${JSON.stringify(payload || {})}\n\n`;
  for (const c of [...clients]) {
    if (!canSee(c, branchId)) continue;
    try { c.res.write(body); } catch { clients.delete(c); }
  }
}

function clientCount() {
  return clients.size;
}

let timer = null;
function startHeartbeat() {
  if (timer) return;
  timer = setInterval(() => {
    for (const c of [...clients]) {
      try { c.res.write(': ping\n\n'); } catch { clients.delete(c); }
    }
  }, HEARTBEAT_MS);
  // これがあるとテストでプロセスが終わらなくなるので、待ち行列には数えさせない
  if (timer.unref) timer.unref();
}
function stopHeartbeat() {
  if (timer) { clearInterval(timer); timer = null; }
}

module.exports = {
  addClient, removeClient, broadcast, clientCount,
  startHeartbeat, stopHeartbeat, HEARTBEAT_MS,
};

   ブラウザ標準の EventSource は Authorization ヘッダを付けられず、

function startStream(){
  STREAM_STOP=false;STREAM_WAIT=1000;connectStream();
}
function stopStream(){
  STREAM_STOP=true;
  if(STREAM_CTRL){try{STREAM_CTRL.abort();}catch(e){}STREAM_CTRL=null;}

async function connectStream(){
  if(STREAM_STOP||!getToken())return;
  const ctrl=new AbortController();STREAM_CTRL=ctrl;
  try{
    const res=await fetch('/api/stream',{
      headers:{Authorization:'Bearer '+getToken()},signal:ctrl.signal});

      if(done)break;

  const choiceGroup=(n,g)=>(g.slots.length===1&&!g.allDay)?choiceRow(n,g.slots[0]):`<div class="cgroup ${IV_OPEN.has(n)?'open':''} ${g.slots.includes(iv.confirmed_datetime)?'sel':''}">

        aria-expanded="${IV_OPEN.has(n)?'true':'false'}"
        onkeydown="if(event.key==='Enter'||event.key===' '){event.preventDefault();toggleChoiceGroup('${id}',${n})}">
        <span class="${rankCls(n)}">${rankLabel(n)}</span><span>${fmtGroupRange(g)}</span>

        <span class="cchev">${ic('chevron')}</span></div>
      <div class="cslots">${g.slots.map(val=>iv.status==='applied'
        ? `<button class="btn sm" onclick="confirmInterview('${id}','${val}')">${hm(val)}〜</button>`
        : `<span class="cslot ${iv.confirmed_datetime===val?'on':''}">${hm(val)}〜</span>`).join('')}</div>
      ${iv.status==='applied'?'<div class="cnote">この中から開始時刻を選んで確定します</div>':''}
    </div>`;
