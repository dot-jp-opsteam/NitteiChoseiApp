# 候補日時UIの刷新 実装計画

> **作業する人へ:** このファイルはタスクごとに上から順に進める。各手順のチェックボックスを埋めながら進めること。

**目標:** 出欠確認をつくるときの「候補の日時」を、ミニカレンダーで日付を選び、ホイールで時刻を決める形にする。

**進め方:** 計算だけをする関数を先に作って検証を通し（Task 1）、そのあと画面を段階的に置き換える。
各タスクの終わりで画面が壊れていない状態を保つこと。

**使うもの:** バニラJS（ES6）、CSS の `scroll-snap`、Node標準ライブラリのみ。

## 全体の決まりごと

- **npm の依存を増やさない。** 外部ライブラリ・CDN・アイコンフォントを追加しない
- **料金が発生する方法は使わない。** 無料枠で完結させる
- `index.html` は ES6（`const`/`let`/テンプレートリテラル可）。`apply.html` は ES5 風なので混ぜない
- 検証は `node tools/test-fmt.mjs` と `node tools/e2e.mjs`。両方が通ること
- コミットのメッセージは日本語。末尾に `Co-Authored-By: Claude Opus 5 <noreply@anthropic.com>` を入れる
- `_backup/` は絶対にコミットしない
- 時刻は `'HH:MM'` の文字列。**上限は `23:59`**（`24:00` は無効）
- 日付は `'YYYY-MM-DD'` の文字列
- 設計書は `_specs/2026-08-08-候補日時UIの刷新-design.md`

- [ ] **手順3: 計算関数を書く**

```javascript
/* ===== 出欠確認の候補日時をつくるための計算 =====
   ここは画面に触らない。tools/test-fmt.mjs が関数を切り出して動かすので、
   HTMLの組み立てを混ぜないこと */

/* 時刻 'HH:MM' と、0時からの分数を行き来する。上限は 23:59（24:00 は無い） */
function reqMins(t){const p=String(t||'').split(':');return (Number(p[0])||0)*60+(Number(p[1])||0);}
function reqHHMM(m){const v=Math.min(Math.max(m,0),1439);return pad(Math.floor(v/60))+':'+pad(v%60);}
function reqShiftDate(date,days){
  const d=new Date(date+'T00:00');
  d.setDate(d.getDate()+days);
  return ymd(d);
}
/* なぞった両端から、間の日付をぜんぶ返す。逆向きになぞられても昇順にそろえる */
function reqDateRange(a,b){
  const from=a<=b?a:b, to=a<=b?b:a;
  const out=[];
  const d=new Date(from+'T00:00'), end=new Date(to+'T00:00');
  while(d<=end){out.push(ymd(d));d.setDate(d.getDate()+1);}
  return out;
}
/* 同じ日をもう1件ふやす。前の終わりから始めて長さは変えない。
   その日に収まらないときは翌日の 0 時へ送る（23:59 が上限なので、
   終わりがちょうど 24:00 になる場合も収まらない扱いにする） */
function copyReqOption(opt){
  const len=Math.max(1,reqMins(opt.t2)-reqMins(opt.t1));
  const from=reqMins(opt.t2);
  if(from+len>1439)return {date:reqShiftDate(opt.date,1),t1:'00:00',t2:reqHHMM(len)};
  return {date:opt.date,t1:reqHHMM(from),t2:reqHHMM(from+len)};
}
/* ホイールで片方を動かしたときの、開始と終わりの組。
   逆転した状態を作らせないよう、長さを保ったまま相手も引き連れて動かす */
function reqApplyTimeChange(opt,field,value){
  const len=Math.max(1,reqMins(opt.t2)-reqMins(opt.t1));
  if(field==='t1'){
    const from=reqMins(value);
    let to=reqMins(opt.t2);
    if(to<=from)to=Math.min(from+len,1439);
    return {t1:reqHHMM(Math.min(from,to-1)),t2:reqHHMM(to)};
  }
  const to=reqMins(value);
  let from=reqMins(opt.t1);
  if(from>=to)from=Math.max(to-len,0);
  return {t1:reqHHMM(from),t2:reqHHMM(Math.max(to,from+1))};
}
/* カレンダーに並べるマス。前の月にはみ出すぶんは null で埋め、
   末尾は行がそろうまで null を足す。dates に入っている日を on にする */
function reqCalendarCells(year,month,dates,today){
  const head=new Date(year,month,1).getDay();
  const days=new Date(year,month+1,0).getDate();
  const cells=[];
  for(let i=0;i<head;i++)cells.push(null);
  for(let d=1;d<=days;d++){
    const date=ymd(new Date(year,month,d));
    cells.push({date:date,day:d,on:dates.indexOf(date)>=0,past:date<today});
  }
  while(cells.length%7)cells.push(null);
  return cells;
}
```

- [ ] **手順2: カレンダーの状態と組み立てを書く**

`index.html` の `reqOptionsHTML`（4622 行のあたり）の直前に足す:

```javascript
/* ===== 候補の日時のミニカレンダー =====
   光っている日は REQFORM.opts から毎回導き出す。別に持たない。
   二重に持つと、コピーで日付がずれたときや行を消したときに片方だけ古くなる */
let REQCAL=null;   // 表示中の月 {y,m}。m は 0〜11
function reqCalInit(){
  if(REQCAL)return;
  const d=new Date();
  REQCAL={y:d.getFullYear(),m:d.getMonth()};
}
function reqPickedDates(){
  const seen=[];
  REQFORM.opts.forEach(o=>{if(o.date&&seen.indexOf(o.date)<0)seen.push(o.date);});
  return seen.sort();
}
function moveReqCal(delta){
  syncReqOptions();
  const d=new Date(REQCAL.y,REQCAL.m+delta,1);
  REQCAL={y:d.getFullYear(),m:d.getMonth()};
  renderRequestForm(true);
}
/* その日を足す、または外す。外すときはその日の候補をぜんぶ消す */
function toggleReqDate(date){
  syncReqOptions();
  const has=REQFORM.opts.some(o=>o.date===date);
  if(has)REQFORM.opts=REQFORM.opts.filter(o=>o.date!==date);
  else REQFORM.opts.push(reqNewOption(date));
  reqSortOptions();
  renderRequestForm(true);
}
/* 新しい日の初期の時刻は、直前の候補と同じにする。1件も無ければ 19:00〜21:00 */
function reqNewOption(date){
  const last=REQFORM.opts[REQFORM.opts.length-1];
  return {date:date,t1:last?last.t1:'19:00',t2:last?last.t2:'21:00'};
}
/* 日付順、同じ日なら開始が早い順。カレンダーの並びと一覧の並びをそろえるため */
function reqSortOptions(){
  REQFORM.opts.sort((a,b)=>a.date===b.date?reqMins(a.t1)-reqMins(b.t1):(a.date<b.date?-1:1));
}
function reqCalendarHTML(){
  reqCalInit();
  const today=ymd(new Date());
  const cells=reqCalendarCells(REQCAL.y,REQCAL.m,reqPickedDates(),today);
  const head='日月火水木金土'.split('').map((w,i)=>
    `<div class="rc-w${i===0?' sun':i===6?' sat':''}">${w}</div>`).join('');
  const body=cells.map(c=>{
    if(!c)return '<div class="rc-d empty"></div>';
    if(c.past)return `<div class="rc-d past">${c.day}</div>`;
    return `<div class="rc-d${c.on?' on':''}" data-date="${c.date}" role="button" tabindex="0"
      aria-pressed="${c.on}" aria-label="${c.date}"
      onpointerdown="reqCalDown(event,'${c.date}')"
      onkeydown="if(event.key==='Enter'||event.key===' '){event.preventDefault();toggleReqDate('${c.date}')}">${c.day}</div>`;
  }).join('');
  return `<div class="reqcal">
      <div class="rc-head">
        <button type="button" class="rc-nav" onclick="moveReqCal(-1)" aria-label="前の月">${ic('back')}</button>
        <b>${REQCAL.y}年${REQCAL.m+1}月</b>
        <button type="button" class="rc-nav" onclick="moveReqCal(1)" aria-label="次の月">${ic('chevron')}</button>
      </div>
      <div class="rc-grid">${head}${body}</div>
    </div>`;
}
/* なぞる仕掛けは Task 4 で入れる。いまは押しただけで切り替える */
function reqCalDown(e,date){
  if(e.button!==undefined&&e.button!==0)return;
  e.preventDefault();
  toggleReqDate(date);
}
```

## Task 4: なぞって選べるようにする

`apply.html` の `applyPointerDown` と同じ考え方。押し始めた時点の状態を控えておき、
指を動かすたびにそこから作り直す。こうすると指を戻したときに選択も戻る。

**ファイル:**
- 変更: `index.html`（Task 3 で足した `reqCalDown` を置き換える）

**この Task が使うもの:** `reqDateRange`（Task 1）、`reqNewOption` / `reqSortOptions`（Task 3）

- [ ] **手順1: `reqCalDown` を置き換える**

```javascript
/* なぞって選ぶ。押し始めた日が未選択なら追加、選択済みなら解除でモードを決め、
   往復してもモードが入れ替わらないようにする。
   毎回「なぞる前の状態」から組み立て直すので、指を戻せば選択も戻る */
let REQDRAG=null;
function reqCalDown(e,date){
  if(e.button!==undefined&&e.button!==0)return;
  e.preventDefault();
  syncReqOptions();
  REQDRAG={
    anchor:date,
    mode:REQFORM.opts.some(o=>o.date===date)?'remove':'add',
    base:REQFORM.opts.map(o=>({date:o.date,t1:o.t1,t2:o.t2})),
  };
  document.body.classList.add('noselect');
  reqCalDragTo(date);
  document.addEventListener('pointermove',reqCalMove);
  document.addEventListener('pointerup',reqCalUp);
  document.addEventListener('pointercancel',reqCalUp);
}
/* 指の下にあるマスの日付を読む。カレンダーの外へ出たときは何もしない */
function reqCalMove(e){
  if(!REQDRAG)return;
  const el=document.elementFromPoint(e.clientX,e.clientY);
  const cell=el&&el.closest?el.closest('.rc-d[data-date]'):null;
  if(cell)reqCalDragTo(cell.dataset.date);
}
function reqCalDragTo(date){
  const d=REQDRAG;
  if(!d)return;
  const range=reqDateRange(d.anchor,date);
  const today=ymd(new Date());
  const targets=range.filter(x=>x>=today);   // 過去日はまたぐだけ
  REQFORM.opts=d.base.map(o=>({date:o.date,t1:o.t1,t2:o.t2}));
  if(d.mode==='add'){
    targets.forEach(x=>{
      if(!REQFORM.opts.some(o=>o.date===x))REQFORM.opts.push(reqNewOption(x));
    });
  }else{
    REQFORM.opts=REQFORM.opts.filter(o=>targets.indexOf(o.date)<0);
  }
  reqSortOptions();
  renderRequestForm(true);
}
function reqCalUp(){
  if(!REQDRAG)return;
  REQDRAG=null;
  document.body.classList.remove('noselect');
  document.removeEventListener('pointermove',reqCalMove);
  document.removeEventListener('pointerup',reqCalUp);
  document.removeEventListener('pointercancel',reqCalUp);
}
```

押した時刻の下に、時と分の2列を開く。`scroll-snap` で止め、止まった位置から時刻を読む。

- [ ] **手順1: 仮の関数を本物に置き換える**

```javascript
/* ===== 時刻のホイール =====
   押した時刻の下に、時と分の列を開く。scroll-snap で止まった位置から値を読む。
   スクロールは細かく発火するので、止まってから読むまで少し待つ */
let REQWHEEL=null;   // 開いている場所 {i, field}
function openReqWheel(i,field){
  const same=REQWHEEL&&REQWHEEL.i===i&&REQWHEEL.field===field;
  REQWHEEL=same?null:{i:i,field:field};
  renderRequestForm(true);
  if(REQWHEEL)requestAnimationFrame(scrollReqWheelIntoPlace);
}
function closeReqWheel(){
  if(!REQWHEEL)return;
  REQWHEEL=null;
  renderRequestForm(true);
}
/* 開いた直後に、いまの時刻の位置まで飛ばす。
   scroll-snap は「近いところで止める」だけなので、初期位置は自分で合わせる */
function scrollReqWheelIntoPlace(){
  if(!REQWHEEL)return;
  const o=REQFORM.opts[REQWHEEL.i];
  if(!o)return;
  const v=o[REQWHEEL.field];
  const h=Number(v.split(':')[0]), m=Number(v.split(':')[1]);
  const hc=document.getElementById('rw_h'), mc=document.getElementById('rw_m');
  if(hc)hc.scrollTop=h*REQ_WHEEL_ROW;
  if(mc)mc.scrollTop=m*REQ_WHEEL_ROW;
}
const REQ_WHEEL_ROW=34;   // style.css の .rw-item の高さと必ずそろえること
/* 止まった位置から時刻を組み立てて控える。逆転は reqApplyTimeChange が直す */
let REQ_WHEEL_TIMER=null;
function onReqWheelScroll(){
  clearTimeout(REQ_WHEEL_TIMER);
  REQ_WHEEL_TIMER=setTimeout(commitReqWheel,120);
}
function commitReqWheel(){
  if(!REQWHEEL)return;
  const o=REQFORM.opts[REQWHEEL.i];
  const hc=document.getElementById('rw_h'), mc=document.getElementById('rw_m');
  if(!o||!hc||!mc)return;
  const h=Math.min(23,Math.max(0,Math.round(hc.scrollTop/REQ_WHEEL_ROW)));
  const m=Math.min(59,Math.max(0,Math.round(mc.scrollTop/REQ_WHEEL_ROW)));
  const next=reqApplyTimeChange(o,REQWHEEL.field,pad(h)+':'+pad(m));
  if(next.t1===o.t1&&next.t2===o.t2)return;   // 変わっていないなら描き直さない
  o.t1=next.t1;o.t2=next.t2;
  reqSortOptions();
  renderRequestForm(true);
  requestAnimationFrame(scrollReqWheelIntoPlace);
}
/* 矢印キーでも動かせるようにする。数字を打つ入力欄は無くしたので、
   キーボードだけで操作する人にはこれが唯一の手段になる。必ず残すこと */
function onReqWheelKey(e,col){
  const step=e.key==='ArrowUp'?-1:e.key==='ArrowDown'?1:
    e.key==='PageUp'?-10:e.key==='PageDown'?10:0;
  if(!step)return;
  e.preventDefault();
  const el=document.getElementById(col==='h'?'rw_h':'rw_m');
  if(!el)return;
  el.scrollTop+=step*REQ_WHEEL_ROW;
  onReqWheelScroll();
}
function reqWheelHTML(i){
  if(!REQWHEEL||REQWHEEL.i!==i)return '';
  const o=REQFORM.opts[i];
  if(!o)return '';
  const v=o[REQWHEEL.field];
  const cur={h:Number(v.split(':')[0]),m:Number(v.split(':')[1])};
  const col=(n,key,label)=>{
    let out='';
    for(let x=0;x<n;x++){
      out+=`<div class="rw-item${x===cur[key]?' on':''}${x%5===0&&key==='m'?' mark':''}">${pad(x)}</div>`;
    }
    return `<div class="rw-col" id="rw_${key}" tabindex="0" role="listbox" aria-label="${label}"
      onscroll="onReqWheelScroll()" onkeydown="onReqWheelKey(event,'${key}')">
      <div class="rw-pad"></div>${out}<div class="rw-pad"></div></div>`;
  };
  return `<div class="reqwheel">
      <div class="rw-cols">${col(24,'h','時')}${col(60,'m','分')}</div>
      <div class="rw-line" aria-hidden="true"></div>
      <button type="button" class="btn ghost sm rw-done" onclick="closeReqWheel()">閉じる</button>
    </div>`;
}
```
